################ 2. Create the 3 containers ################
# Frontend = Nginx, attached to frontend-net
$ docker run -d --name frontend --network frontend-net -p 8085:80 nginx:alpine
f55e31d2c7b4aac857eaf6209045689537431cbeccfff3cafbd38b9c9199f2e4

# Database = MySQL, attached to db-net only
$ docker run -d --name database --network db-net -e MYSQL_ROOT_PASSWORD=rootpw -e MYSQL_DATABASE=appdb mysql:8.4
62f2b98b76d59d1ef787bf9039907e5ec2e9070831999ecc3575d51fb1d159a9

# Backend = Alpine, attached to frontend-net FIRST ...
$ docker run -d --name backend --network frontend-net alpine sh -c "apk add --no-cache busybox-extras >/dev/null && mkdir -p /www && echo backend-tier-OK > /www/index.html && httpd -f -p 5000 -h /www"

################ 5. Check connectivity between the containers ################

--- 5a. frontend -> backend (SAME network: frontend-net) => expect SUCCESS ---
$ docker exec frontend wget -qO- http://backend:5000/
backend-tier-OK
exit=0

--- 5b. backend -> database (SAME network: db-net) => expect SUCCESS ---
$ docker exec backend getent hosts database     # DNS resolution over db-net
172.20.0.2        database  database
$ docker exec backend nc -z -w 3 database 3306 && echo "TCP 3306 reachable"
TCP 3306 reachable

--- 5c. frontend -> database (NO shared network) => expect FAILURE (isolation) ---
$ docker exec frontend getent hosts database
exit=2
$ docker exec frontend nc -z -w 3 database 3306
nc: bad address 'database'
exit=1 (non-zero = unreachable, which is the point)

--- 5d. a container on the 3rd network (backend-net) reaches NOTHING => isolation ---
$ docker run -d --name isolated --network backend-net alpine sleep 300
b6af85b6a15cc75d2443ec8c67f2385c352070938eddbcc6f8b7c7a6dafde544
$ docker exec isolated getent hosts frontend
  -> exit=2 (2 = name does not resolve; no shared network)
$ docker exec isolated getent hosts backend
  -> exit=2 (2 = name does not resolve; no shared network)
$ docker exec isolated getent hosts database
  -> exit=2 (2 = name does not resolve; no shared network)

--- 5e. proof the frontend itself serves from the host ---
