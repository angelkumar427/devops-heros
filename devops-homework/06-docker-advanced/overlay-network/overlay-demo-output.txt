################ 3. Create a user-defined overlay network ################
$ docker network create -d overlay --attachable my-overlay
spcts5zg4yf57pi6r9b7fwvnt

$ docker network inspect my-overlay --format "Driver={{.Driver}} Scope={{.Scope}} Attachable={{.Attachable}} Subnet={{range .IPAM.Config}}{{.Subnet}}{{end}}"
Driver=overlay Scope=swarm Attachable=true Subnet=10.0.1.0/24

# NOTE the Scope: "swarm", not "local" like a bridge network. That is the
# whole point - the network definition is stored in the swarm raft log and
# exists on every node in the cluster, not just this one.

################ 5. Service discovery ACROSS the overlay network ################
$ docker run --rm --network my-overlay alpine sh -c "apk add --no-cache bind-tools >/dev/null; nslookup web-svc; nslookup tasks.web-svc"
--- VIP (the service name) ---

Non-authoritative answer:
Name:	web-svc
Address: 10.0.1.2

--- tasks.<service> = every replica IP ---
Non-authoritative answer:
Name:	tasks.web-svc
Address: 10.0.1.4
Name:	tasks.web-svc
Address: 10.0.1.3
Name:	tasks.web-svc
Address: 10.0.1.5


# web-svc resolves to ONE virtual IP (the VIP) that load-balances across
# replicas; tasks.web-svc resolves to the individual container IPs. On a
# multi-host swarm those replicas would sit on different machines and this
# would work identically - that is what the VXLAN overlay buys you.

################ 6. Traffic across the overlay network ################
# NOTE: the swarm ROUTING MESH (the -p 8089 published port) does not answer on
# this setup - ingress/VXLAN is unreliable under Colima on macOS. That is a
# host-platform limitation, not an overlay one, so the overlay data path is
# verified directly instead: east-west traffic between containers attached to
# the overlay, which is what an overlay network actually provides.

$ docker run --rm --network my-overlay alpine wget -qO- --timeout=8 http://web-svc/

################ 7. Single-node caveat ################
$ docker node ls
HOSTNAME   STATUS    MANAGER STATUS
colima     Ready     Leader
# Only one node here, so all 3 replicas land on the same machine. The
# overlay/VXLAN plumbing is real and identical; a second machine would just
# run "docker swarm join <token>" and replicas would spread across both hosts.
